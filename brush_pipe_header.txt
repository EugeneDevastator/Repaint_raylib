// ─── PaintManager ─────────────────────────────────────────────────────────

struct PaintManager {
    std::unordered_map<int, UserExecutionState>      userStates;
    std::unordered_map<int, std::deque<DrawCommand>> commandQueues;
    std::unordered_map<int, IStrokeEngine*>          strokeEngines;

    // called by local input / network receive
    void EnqueueCommand(int userId, DrawCommand cmd);

    // call each frame
    void Tick();
};

void PaintManager::Tick() {
    for (auto& [userId, queue] : commandQueues) {
        if (!segmentDrawer->CanDraw()) break;
        if (queue.empty()) continue;

        auto& state = userStates[userId];
        auto& cmd   = queue.front();

        std::visit(overloaded{
            [&](StartStrokeCommand& c) {
                state.lastSegmentDelta = {};
                state.currentStrokeId  = c.strokeId;
                strokeEngines[userId]->StartStroke(c.targetQuadId);
            },
            [&](PackedStrokeSegment& seg) {
                state.lastSegmentDelta =
                    segmentBuilder->ComputeNextDelta(seg, state.lastSegmentDelta);
                auto dabbed = segmentBuilder->Build(seg, state.lastSegmentDelta, state);
                segmentDrawer->Draw(dabbed);
            },
            [&](DrawRectCommand& c) {
                rectDrawer->Draw(c);  // color already snapshotted
            },
            [&](LayerTransformCommand& c) {
                layerSystem->Apply(c);
            },
            [&](BrushUserModelPacked& m) {
                state.brushModel = m;
            },
        }, cmd);

        queue.pop_front();
    }
}
